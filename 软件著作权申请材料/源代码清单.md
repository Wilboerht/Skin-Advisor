```typescript
export async function analyzeImages(
  images: VisionImage[],
  context: AnalysisContext,
  retries: number = 0
): Promise<FaceAnalysisResult> {
  // 1. 构建多模态提示词（包含4张照片）
  const messages = buildVisionPrompt(images, context);
  
  // 2. 调用AI API（OpenAI/Qwen/Claude轮询）
  const client = createOpenAIClient(provider);
  const response = await client.chat.completions.create({
    model: getVisionModel(provider),
```

```typescript
// 核心创新点：MediaPipe Face Mesh 468点检测
// 功能：单例模式初始化FaceLandmarker，本地化运行

let faceLandmarker: FaceLandmarker | null = null;
let initStatus: InitStatus = "idle";
let initPromise: Promise<FaceLandmarker | null> | null = null;

export async function initFaceLandmarker(): Promise<FaceLandmarker | null> {
  // 单例模式：复用已初始化的实例
  if (initStatus === "ready" && faceLandmarker) {
    return faceLandmarker;
  }
  
  // 并发安全：多次调用返回同一个Promise
  if (initStatus === "loading" && initPromise) {
    return initPromise;
  }
  
  // 初始化流程
  initStatus = "loading";
  
  initPromise = (async () => {
    // 1. 加载WASM运行时（本地public目录）
    const vision = await FilesetResolver.forVisionTasks(WASM_PATH);
    
    // 2. 加载模型文件
    faceLandmarker = await FaceLandmarker.createFromModelPath(
      vision,
      MODEL_PATH // face_landmarker.task
    );
    
    initStatus = "ready";
    return faceLandmarker;
  })();
  
  return initPromise;
}
```

**创新性说明**：
- Web端成功集成MediaPipe Face Mesh
- 本地化WASM部署，适配国内网络
- 单例模式避免重复初始化

```typescript
export type ZoneKey = "forehead" | "tZone" | "leftCheek" | "rightCheek" | "eyeArea" | "jawline";
```

```typescript
// 核心创新点：环境感知的动态护肤推荐
// 功能：结合天气API调整护肤方案

export async function getClimateByRegion(region: string): Promise<ClimateData> {
  const latLon = REGION_COORDINATES[region] || REGION_COORDINATES["北京"];
  
  // 调用Open-Meteo天气API
  const url = `https://api.open-meteo.com/v1/forecast?latitude=${latLon.lat}&longitude=${latLon.lon}&current=uv_index,relative_humidity_2m,weather_code,surface_pressure,cloud_cover&daily=uv_index_max,precipitation_probability_sum&forecast_days=1&timezone=auto`;
  
  const response = await fetch(url);
  const data = await response.json();
  
  return {
    uvIndex: data.current.uv_index,
    humidity: data.current.relative_humidity_2m,
    weatherCode: data.current.weather_code,
    pressure: data.current.surface_pressure,
    cloudCover: data.current.cloud_cover,
    
    // 预测明天
    tomorrow: {
      uvIndexMax: data.daily.uv_index_max[0],
      precipitationProbability: data.daily.precipitation_probability_sum[0],
    },
  };
}

// 动态调整护肤建议
export function adjustRecommendations(climate: ClimateData, recommendations: string[]): string[] {
  const adjusted = [...recommendations];
  
  // UV指数高 - 强化防晒
  if (climate.uvIndex > 8) {
    adjusted.push("[!] 当前紫外线指数极高（>8），建议使用SPF50+防晒霜，每2小时补涂");
    adjusted.push("[提示] 戴帽子和太阳镜进行物理防晒");
  } else if (climate.uvIndex > 5) {
    adjusted.push("[i] 当前紫外线指数中等（5-8），建议使用SPF30+以上防晒霜");
  } else if (climate.uvIndex < 3) {
    adjusted.push("[!] 当前紫外线指数低（<3），可适当减少防晒频次");
  }
  
  // 湿度低 - 加强保湿
  if (climate.humidity < 30) {
    adjusted.push("[补水] 当前空气湿度较低（<30%），建议使用保湿精华+面霜锁水");
    adjusted.push("[提示] 睡前可使用保湿面膜加强补水");
  } else if (climate.humidity > 80) {
    adjusted.push("/cloud 当前空气湿度较高（>80%），选择轻薄型保湿产品");
  }
  
  // 空气差 - 增加清洁
  if (climate.weatherCode >= 51 && climate.weatherCode <= 67) { // 雾/霾
    adjusted.push("-mask 空气质量较差，建议使用温和洁面产品清洁面部");
  }
  
  return adjusted;
}
```

**创新性说明**：
- 实时天气数据接入
- 动态调整护肤建议
- 超越竞品的静态推荐系统
